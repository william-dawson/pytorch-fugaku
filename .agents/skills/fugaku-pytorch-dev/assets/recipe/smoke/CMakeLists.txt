cmake_minimum_required(VERSION 3.27)
project(torch_fugaku_smoke CXX)
find_package(Torch REQUIRED)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_executable(torch-mpi-smoke main.cpp)
target_compile_features(torch-mpi-smoke PRIVATE cxx_std_17)
target_compile_definitions(torch-mpi-smoke PRIVATE USE_C10D_MPI OMPI_SKIP_MPICXX=1)
target_link_libraries(torch-mpi-smoke PRIVATE ${TORCH_LIBRARIES} MPI::MPI_CXX OpenMP::OpenMP_CXX)
target_link_options(torch-mpi-smoke PRIVATE "LINKER:-rpath-link,${FUGAKU_FJ}/lib64")
